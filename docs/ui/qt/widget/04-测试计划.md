# Qt Widget 桌面客户端 - 测试计划

## 1. 测试概述

### 1.1 测试目标

本测试计划旨在确保 Qt Widget 桌面客户端的功能正确性、稳定性和用户体验。测试覆盖以下方面：

1. **功能测试**：验证所有功能模块按需求正确实现
2. **API 集成测试**：验证与后端服务的通信正确性
3. **UI 测试**：验证 Widget 界面显示和交互
4. **性能测试**：验证响应速度和资源占用
5. **兼容性测试**：验证跨平台兼容性（Windows/Linux/macOS）

### 1.2 测试范围

| 模块 | 测试类型 | 优先级 |
|------|----------|--------|
| API 客户端 | 单元/集成 | P0 |
| 认证服务 | 单元/集成 | P0 |
| 文件服务 | 单元/集成 | P0 |
| 上传服务 | 单元/集成 | P0 |
| 下载服务 | 单元/集成 | P0 |
| Token 存储 | 单元 | P0 |
| Model 类 | 单元 | P0 |
| Controller 层 | 单元/集成 | P0 |
| 文件操作流程 | 集成 | P0 |
| 传输管理 | 集成 | P0 |
| Widget 组件 | UI 自动化 | P1 |
| 回收站功能 | 集成 | P1 |
| 分享功能 | 集成 | P1 |
| 快捷键 | 手动 | P1 |
| 跨平台兼容 | 手动 | P1 |

- P0 模块必须有完整的单元测试和集成测试

- 关键路径必须 100% 测试覆盖

```cpp
// tests/api/ApiClientTest.cpp
#include <QTest>
#include <QSignalSpy>
#include "api/ApiClient.h"
#include "mock/MockNetworkReply.h"

class ApiClientTest : public QObject {
    Q_OBJECT

private:
    api::ApiClient* m_client;
    QNetworkAccessManager* m_mockManager;

private slots:
    void initTestCase() {
        // 全局初始化
        qRegisterMetaType<QNetworkReply::NetworkError>();
    }

    void cleanupTestCase() {
        // 全局清理
    }

    void init() {
        // 每个测试前创建新实例
        m_mockManager = new QNetworkAccessManager(this);
        m_client = new api::ApiClient(this);
    }

    void cleanup() {
        // 每个测试后清理
        delete m_client;
        delete m_mockManager;
    }

    // 测试 URL 构建
    void testBuildRequest_correctUrl() {
        m_client->setBaseUrl("http://localhost:8080");
        
        QNetworkRequest request = m_client->buildRequest("/api/file/list");
        
        QCOMPARE(request.url().toString(), QString("http://localhost:8080/api/file/list"));
        QCOMPARE(request.header(QNetworkRequest::ContentTypeHeader).toString(), 
                 QString("application/json"));
    }

    // 测试认证头注入
    void testBuildRequest_withAuthToken() {
        m_client->setBaseUrl("http://localhost:8080");
        m_client->setTokenProvider([]() { return QString("test_token_123"); });
        
        QNetworkRequest request = m_client->buildRequest("/api/user/profile");
        
        QCOMPARE(request.rawHeader("Authorization"), QByteArray("Bearer test_token_123"));
    }

    // 数据驱动测试 - 响应解析
    void testParseResponse_data() {
        QTest::addColumn<QString>("jsonData");
        QTest::addColumn<bool>("expectedSuccess");
        QTest::addColumn<int>("expectedCode");
        
        QTest::newRow("success response")
            << R"({"code":0,"message":"success","data":{"id":1}})"
            << true << 0;
        
        QTest::newRow("error response")
            << R"({"code":40001,"message":"参数错误","data":null})"
            << false << 40001;
        
        QTest::newRow("invalid json")
            << "not valid json"
            << false << -1;
    }

    void testParseResponse() {
        QFETCH(QString, jsonData);
        QFETCH(bool, expectedSuccess);
        QFETCH(int, expectedCode);
        
        auto response = api::ApiResponse::parse(jsonData.toUtf8());
        
        if (expectedCode == -1) {
            QVERIFY(!response.has_value());
        } else {
            QVERIFY(response.has_value());
            QCOMPARE(response->isSuccess(), expectedSuccess);
            QCOMPARE(response->code, expectedCode);
        }
    }

    // 测试网络错误处理
    void testNetworkError_returnsError() {
        MockNetworkReply* mockReply = new MockNetworkReply();
        mockReply->setError(QNetworkReply::ConnectionRefusedError, "Connection refused");
        
        QSignalSpy errorSpy(m_client, &api::ApiClient::networkError);
        
        // 模拟错误响应
        m_client->handleReplyFinished(mockReply);
        
        QCOMPARE(errorSpy.count(), 1);
        QVERIFY(errorSpy.at(0).at(0).toString().contains("Connection refused"));
        
        mockReply->deleteLater();
    }

    // 测试超时处理
    void testTimeout_returnsError() {
        m_client->setTimeout(100); // 100ms 超时
        
        QSignalSpy errorSpy(m_client, &api::ApiClient::networkError);
        
        // 模拟超时（这里简化处理）
        QTimer::singleShot(150, [this]() {
            // 验证超时触发
        });
        
        QTRY_COMPARE_WITH_TIMEOUT(errorSpy.count(), 1, 200);
    }
};

QTEST_MAIN(ApiClientTest)
#include "ApiClientTest.moc"
```

```cpp
// tests/services/AuthServiceTest.cpp
#include <QTest>
#include <QSignalSpy>
#include "services/AuthService.h"
#include "mock/MockApiClient.h"
#include "mock/MockTokenStore.h"

class AuthServiceTest : public QObject {
    Q_OBJECT

private:
    MockApiClient* m_mockClient;
    MockTokenStore* m_mockStore;
    services::AuthService* m_service;

private slots:
    void init() {
        m_mockClient = new MockApiClient(this);
        m_mockStore = new MockTokenStore(this);
        m_service = new services::AuthService(m_mockClient, m_mockStore, this);
    }

    void cleanup() {
        delete m_service;
        delete m_mockStore;
        delete m_mockClient;
    }

    // 测试登录成功
    void testLogin_success() {
        // 设置模拟响应
        m_mockClient->setMockResponse("/api/auth/login", 
            QJsonDocument::fromJson(R"({
                "code": 0,
                "message": "success",
                "data": {
                    "access_token": "test_access_token",
                    "refresh_token": "test_refresh_token",
                    "expires_in": 7200,
                    "user": {
                        "id": 1,
                        "username": "testuser",
                        "storage_used": 1024,
                        "storage_quota": 10737418240
                    }
                }
            })"));
        
        QSignalSpy authSpy(m_service, &services::AuthService::authenticated);
        
        auto future = m_service->login("testuser", "password123");
        future.waitForFinished();
        
        auto result = future.result();
        
        QVERIFY(std::holds_alternative<services::LoginResult>(result));
        auto& loginResult = std::get<services::LoginResult>(result);
        QCOMPARE(loginResult.accessToken, QString("test_access_token"));
        QCOMPARE(loginResult.user.username, QString("testuser"));
        
        QCOMPARE(authSpy.count(), 1);
        QVERIFY(m_service->isAuthenticated());
    }

    // 测试登录失败 - 密码错误
    void testLogin_wrongPassword() {
        m_mockClient->setMockResponse("/api/auth/login",
            QJsonDocument::fromJson(R"({
                "code": 40004,
                "message": "密码错误",
                "data": null
            })"));
        
        QSignalSpy failSpy(m_service, &services::AuthService::authenticationFailed);
        
        auto future = m_service->login("testuser", "wrongpassword");
        future.waitForFinished();
        
        auto result = future.result();
        
        QVERIFY(std::holds_alternative<utils::ErrorInfo>(result));
        QVERIFY(!m_service->isAuthenticated());
        QCOMPARE(failSpy.count(), 1);
    }

    // 测试 Token 刷新
    void testRefreshToken_success() {
        m_mockClient->setMockResponse("/api/auth/refresh",
            QJsonDocument::fromJson(R"({
                "code": 0,
                "data": {
                    "access_token": "new_access_token",
                    "refresh_token": "new_refresh_token",
                    "expires_in": 7200
                }
            })"));
        
        m_mockStore->setRefreshToken("old_refresh_token", 
                                      QDateTime::currentDateTime().addDays(1));
        
        QSignalSpy refreshSpy(m_service, &services::AuthService::tokenRefreshed);
        
        auto future = m_service->refreshToken();
        future.waitForFinished();
        
        QVERIFY(m_service->hasValidToken());
        QCOMPARE(refreshSpy.count(), 1);
    }

    // 测试登出
    void testLogout_clearsToken() {
        // 先登录
        m_mockClient->setMockResponse("/api/auth/login",
            QJsonDocument::fromJson(R"({
                "code": 0,
                "data": {
                    "access_token": "test_token",
                    "refresh_token": "test_refresh",
                    "expires_in": 7200,
                    "user": {"id": 1, "username": "test"}
                }
            })"));
        
        m_service->login("test", "pass").waitForFinished();
        QVERIFY(m_service->isAuthenticated());
        
        // 登出
        QSignalSpy logoutSpy(m_service, &services::AuthService::loggedOut);
        
        m_service->logout().waitForFinished();
        
        QVERIFY(!m_service->isAuthenticated());
        QVERIFY(m_mockStore->accessToken().isEmpty());
        QCOMPARE(logoutSpy.count(), 1);
    }
};

QTEST_MAIN(AuthServiceTest)
#include "AuthServiceTest.moc"
```

```cpp
#include <QTest>
#include <QSignalSpy>
#include <QTemporaryFile>

class UploadServiceTest : public QObject {

private:
    MockApiClient* m_mockClient;

        m_mockClient = new MockApiClient(this);
        m_service = new services::UploadService(m_mockClient, this);

        delete m_mockClient;
    }

    void testInstantUpload() {

            QJsonDocument::fromJson(R"({

                    "instant_upload": true,
                    "file": {
                        "id": 123,
                        "name": "test.dat",
                        "size": 102400
                    }
                }
            })"));
        
        QSignalSpy completedSpy(m_service, &services::UploadService::taskCompleted);
        
        auto task = m_service->createTask(m_testFilePath, 0);

        QTRY_COMPARE_WITH_TIMEOUT(task->status(), services::UploadStatus::Completed, 2000);
        QCOMPARE(completedSpy.count(), 1);

    void testChunkedUpload() {
        // 模拟初始化响应
        m_mockClient->setMockResponse("/api/file/upload/init",
            QJsonDocument::fromJson(R"({
                "code": 0,
                "data": {
                    "upload_id": "test-upload-id",
                    "chunk_size": 5242880,
                    "total_chunks": 1
                }
            })"));
        
        // 模拟分片上传响应
        m_mockClient->setMockResponse("/api/file/upload/chunk",
            QJsonDocument::fromJson(R"({
                "code": 0,
                "data": {"chunk_number": 1}
            })"));
        
        // 模拟完成响应
        m_mockClient->setMockResponse("/api/file/upload/complete",
            QJsonDocument::fromJson(R"({
                "code": 0,
                "data": {
                    "id": 456,
                    "name": "test.dat",
                    "size": 102400
                }
            })"));
        
        QList<double> progressValues;
        connect(m_service, &services::UploadService::taskProgressChanged,
                this, [&](auto, qint64 sent, qint64 total) {
                    progressValues.append((double)sent / total);
                });
        
        auto task = m_service->createTask(m_testFilePath, 0);
        m_service->startTask(task);
        
        QTRY_COMPARE_WITH_TIMEOUT(task->status(), services::UploadStatus::Completed, 3000);
        
        // 验证进度更新
        QVERIFY(progressValues.size() > 0);
        QCOMPARE(progressValues.last(), 1.0);
    }

        m_service->startTask(task);

        task->pause();
        
        QCOMPARE(task->status(), services::UploadStatus::Paused);
        
        task->resume();

                "code": 40009,

                "data": null
            })"));

        m_service->startTask(task);
        
        QTRY_COMPARE_WITH_TIMEOUT(task->status(), services::UploadStatus::Failed, 2000);
        QVERIFY(task->errorMessage().contains("存储空间"));
    }
```

```cpp
// tests/models/FileListModelTest.cpp
#include <QTest>
#include <QSignalSpy>
#include "models/FileListModel.h"

class FileListModelTest : public QObject {
    Q_OBJECT

private:
    models::FileListModel* m_model;

private slots:
    void init() {
        m_model = new models::FileListModel(this);
    }

    void cleanup() {
        delete m_model;
    }

    // 测试设置文件列表
    void testSetFiles() {
        QList<models::FileInfo> files;
        models::FileInfo file1;
        file1.id = 1;
        file1.name = "test.txt";
        file1.type = models::FileType::File;
        file1.size = 1024;
        files.append(file1);
        
        QList<models::FolderInfo> folders;
        models::FolderInfo folder1;
        folder1.id = 2;
        folder1.name = "Documents";
        folders.append(folder1);
        
        QSignalSpy countSpy(m_model, &models::FileListModel::countChanged);
        
        m_model->setFiles(files, folders);
        
        QCOMPARE(m_model->rowCount(), 2);
        QCOMPARE(countSpy.count(), 1);
        
        // 验证文件夹在前面
        QModelIndex firstIndex = m_model->index(0);
        QVERIFY(m_model->data(firstIndex, models::FileListModel::IsFolderRole).toBool());
        QCOMPARE(m_model->data(firstIndex, models::FileListModel::NameRole).toString(),
                 QString("Documents"));
    }

    // 测试角色数据
    void testData_roles() {
        models::FileInfo file;
        file.id = 100;
        file.name = "document.pdf";
        file.size = 2048;
        file.type = models::FileType::File;
        
        m_model->setFiles({file}, {});
        
        QModelIndex index = m_model->index(0);
        
        QCOMPARE(m_model->data(index, models::FileListModel::IdRole).toULongLong(), 
                 quint64(100));
        QCOMPARE(m_model->data(index, models::FileListModel::NameRole).toString(),
                 QString("document.pdf"));
        QCOMPARE(m_model->data(index, models::FileListModel::SizeRole).toLongLong(),
                 qint64(2048));
        QVERIFY(!m_model->data(index, models::FileListModel::IsFolderRole).toBool());
    }

    // 测试排序
    void testSortByName() {
        QList<models::FileInfo> files;
        for (int i = 3; i >= 1; --i) {
```

```cmake
function(add_disk_test NAME)
    set(SOURCES ${ARGN})
    
    add_executable(${NAME} ${SOURCES})
    
    target_link_libraries(${NAME} PRIVATE
        disk-core
        Qt6::Test
        mock
    )
    
    target_include_directories(${NAME} PRIVATE
        ${CMAKE_SOURCE_DIR}/src
        ${CMAKE_CURRENT_SOURCE_DIR}/mock
    )
    
    # 添加到 CTest
    add_test(NAME ${NAME} COMMAND ${NAME})
    
    # 设置输出目录
    set_target_properties(${NAME} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/tests
    )
endfunction()
```

### 10.3 持续集成配置

```yaml
# .github/workflows/test.yml
name: Qt Widget Tests

on:
  push:
    branches: [main, develop]
    paths:
      - 'src/ui/qt/widget/**'
      - 'tests/**'
  pull_request:
    branches: [main]

jobs:
  test:
    runs-on: ${{ matrix.os }}
    strategy:
      matrix:
        os: [ubuntu-22.04, windows-2022, macos-13]
        qt_version: ['6.5.0']

    steps:
      - uses: actions/checkout@v4
      
      - name: Install Qt
        uses: jurplel/install-qt-action@v3
        with:
          version: ${{ matrix.qt_version }}
          modules: 'qtbase'
          
      - name: Configure
        run: |
          cmake -B build -DCMAKE_BUILD_TYPE=Debug \
                -DBUILD_TESTS=ON \
                -DCMAKE_CXX_FLAGS="--coverage"
        
      - name: Build
        run: cmake --build build --parallel
        
      - name: Run Tests
        run: ctest --test-dir build --output-on-failure -T Test
        
      - name: Generate Coverage
        if: runner.os == 'Linux'
        run: |
          lcov --capture --directory build --output-file coverage.info
          lcov --remove coverage.info '/usr/*' --output-file coverage.info
          
      - name: Upload Coverage
        if: runner.os == 'Linux'
        uses: codecov/codecov-action@v3
        with:
          files: coverage.info
          
      - name: Upload Test Results
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: test-results-${{ matrix.os }}
          path: build/Testing/
```
