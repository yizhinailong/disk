# TUI 客户端 - 测试计划

## 1. 测试概述

### 1.1 测试目标

本测试计划旨在确保 TUI（终端用户界面）网盘客户端的功能正确性、稳定性和用户体验。测试覆盖以下方面：

1. **功能测试**：验证所有功能模块按需求正确实现
2. **API 集成测试**：验证与后端服务的通信正确性
3. **UI 交互测试**：验证终端界面的响应和显示
4. **边界条件测试**：验证异常输入和网络错误的处理
**单元测试范围说明**：TUI 的页面渲染与交互（View 层）不做单元测试；单元测试聚焦 API 客户端、配置管理、Token 存储等非 UI 逻辑模块。

### 1.2 测试范围

| 模块 | 测试类型 | 优先级 |
|------|----------|--------|
| 认证模块 | 单元/集成 | P0 |
| API 客户端 | 单元/集成 | P0 |
| 配置管理 | 单元 | P0 |
| Token 存储 | 单元/集成 | P0 |
| 文件操作 | 集成 | P0 |

func TestAPIClient_Login_Success(t *testing.T) {
    // 创建 Mock 服务器
    server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        assert.Equal(t, "/api/auth/login", r.URL.Path)
        assert.Equal(t, "application/json", r.Header.Get("Content-Type"))
        
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(http.StatusOK)
        json.NewEncoder(w).Encode(map[string]interface{}{
            "code": 0,
            "data": map[string]interface{}{
                "access_token":  "test_access_token",
                "refresh_token": "test_refresh_token",
                "expires_in":    7200,
            },
        })
    }))
    defer server.Close()
    
    // 测试 API 客户端
    client := api.NewClient(server.URL)
    resp, err := client.Login(context.Background(), "testuser", "password123")
    
    require.NoError(t, err)
    assert.Equal(t, "test_access_token", resp.AccessToken)
}
```

`

// MockTokenStorage 实现 TokenStorage 接口
type MockTokenStorage struct {
    mock.Mock
}

func (m *MockTokenStorage) Save(token *Token) error {
    args := m.Called(token)
    return args.Error(0)
}

func (m *MockTokenStorage) Load() (*Token, error) {
    args := m.Called()
    if args.Get(0) == nil {
        return nil, args.Error(1)
    }
    return args.Get(0).(*Token), args.Error(1)
}

func (m *MockTokenStorage) Clear() error {
    args := m.Called()
    return args.Error(0)
}

func TestTokenManager_GetValidToken_Expired(t *testing.T) {
    mockStorage := new(MockTokenStorage)
    manager := token.NewManager(mockStorage)
    
    // 设置已过期的 Token
    expiredToken := &token.Token{
        AccessToken:  "expired_access",
        RefreshToken: "valid_refresh",
        ExpiresAt:    time.Now().Add(-1 * time.Hour),
    }
    mockStorage.On("Load").Return(expiredToken, nil)
    
    // 测试 Token 刷新逻辑
    validToken, err := manager.GetValidToken(context.Background())
    
    require.NoError(t, err)
    assert.NotEqual(t, "expired_access", validToken.AccessToken)
    mockStorage.AssertExpectations(t)
}
```

#### 7.3.2 自动化回归

```bash
# CI/CD 回归测试脚本
#!/bin/bash
# scripts/regression_test.sh

echo "Starting regression tests..."

# 1. 单元测试
go test ./... -v -count=1
if [ $? -ne 0 ]; then
    echo "Unit tests failed!"
    exit 1
fi

# 2. 集成测试
go test ./tests/integration/... -v -count=1
if [ $? -ne 0 ]; then
    echo "Integration tests failed!"
    exit 1
fi

# 3. 覆盖率检查
go test -coverprofile=coverage.out ./...
coverage=$(go tool cover -func=coverage.out | grep total | awk '{print $3}' | sed 's/%//')
if [ $(echo "$coverage < 70" | bc) -eq 1 ]; then
    echo "Coverage $coverage% is below 70%!"
    exit 1
fi

echo "All regression tests passed!"
```
